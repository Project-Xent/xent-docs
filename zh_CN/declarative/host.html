6. 文件对话框、拖放、主题切换与后台线程¶

桌面应用经常需要与操作系统外壳协作——打开原生文件选择器、接收资源管理器拖入的文件、读写剪贴板、切换深浅色主题，以及在后台线程运行耗时任务。本章介绍如何将这些能力自然融入声明式 MVU 循环。

通过 on_start 与 on_stop 获取 FluxApp *¶

flux_run 会自动创建并管理 FluxApp * 实例。在 FluxAppCfg 上配置 on_start 、 on_stop 与 user ，即可在窗口生命周期内将 FluxApp * 指针保存在 Model 中：

typedef struct Model {
    FluxApp *app;
    char    *chosen_folder;
    bool     dark_mode;
} Model;

static void app_start(FluxApp *app, void *user) {
    Model *m = user;
    m->app   = app;
}

static void app_stop(FluxApp *app, void *user) {
    (void) app;
    Model *m = user;
    // Join any background threads before clearing m->app
    m->app = NULL;
}

int WINAPI WinMain(HINSTANCE hInst, HINSTANCE hPrev, LPSTR cmdLine, int showCmd) {
    (void) hInst; (void) hPrev; (void) cmdLine; (void) showCmd;
    flux_controls_all();

    static Model model;
    FluxAppCfg cfg = {
        .title    = L"Shell Integration",
        .width    = 720,
        .height   = 480,
        .backdrop = FLUX_BACKDROP_MICA,
        .on_start = app_start,
        .on_stop  = app_stop,
        .user     = &model,
    };
    return flux_run(&cfg, &model, update, view);
}

原生文件与文件夹选择对话框¶

引入 <fluxent/flux_filedialog.h> ，在用户点击“浏览…”按钮时于 update() 内调用 flux_filedialog_post ，即可弹出原生 Windows 文件对话框并将选择结果作为 XtkMsg 投递回 update() ：

static FluxFileFilter const kDocFilters[] = {
    {.name = "Markdown & Text", .spec = "*.md;*.txt"},
    {.name = "All Files",       .spec = "*.*"},
};

// Inside update():
case MSG_BROWSE_OPEN: {
    FluxFileDialogDesc desc = {
        .kind         = FLUX_FILEDIALOG_OPEN, // or FLUX_FILEDIALOG_SAVE / FLUX_FILEDIALOG_FOLDER
        .title        = "Open Document",
        .filters      = kDocFilters,
        .filter_count = 2,
        .multiple     = false,
    };
    flux_filedialog_post(m->app, &desc, xtk_msg(MSG_FILE_PICKED));
    break;
}
case MSG_FILE_PICKED:
    // msg.b is true when the user confirmed; msg.ptr holds UTF-8 path(s), one per line
    if (msg.b && msg.ptr) {
        free(m->chosen_folder);
        m->chosen_folder = strdup((char const *) msg.ptr);
    }
    break;

OLE 拖放与剪贴板¶

放置目标与拖拽源（ xtk_dragdrop ）¶

使用 xtk_dragdrop(ui, desc, child) 包裹任意容器或卡片，即可接收从资源管理器及其他程序拖入的文件或文本，或将该元素设为可向外拖出的拖拽源：

// Drop target:
xtk_dragdrop(
    ui,
    (XtkDragDropDesc) {
        .drop       = XTK_DROP_COPY,
        .hover      = xtk_msg(MSG_DROP_HOVER), // msg.b = pointer is hovering over target
        .drop_files = xtk_msg(MSG_DROP_FILES), // msg.ptr = newline-separated UTF-8 paths
        .drop_text  = xtk_msg(MSG_DROP_TEXT),  // msg.ptr = dropped UTF-8 text
    },
    xtk_card(
        ui, (XtkStackDesc) {.padding = {24, 24, 24, 24}},
        xtk_kids(xtk_text(ui, m->hovering ? "Release to import" : "Drop files here", (XtkTextDesc) {0}))
    )
)

// Drag source:
xtk_dragdrop(
    ui,
    (XtkDragDropDesc) {
        .drag_text = "Snippet from Xent", // or .drag_files = "C:\\path\\file.txt"
        .dragged   = xtk_msg(MSG_DRAG_FINISHED),
    },
    xtk_card(ui, (XtkStackDesc) {0}, xtk_kids(xtk_text(ui, "Drag me", (XtkTextDesc) {0})))
)

与 TextBox 相同，处理 MSG_DROP_FILES 或 MSG_DROP_TEXT 时请在 update() 内用 strdup 拷贝 (char const *) msg.ptr 。

剪贴板读写¶

在 update() 中引入 <fluxent/flux_data.h> 即可读写系统剪贴板：

// Copy UTF-8 text to clipboard:
flux_clipboard_settext(text, strlen(text));

// Paste UTF-8 text from clipboard (caller frees the returned buffer):
char *pasted = flux_clipboard_text();
if (pasted) {
    free(m->notes);
    m->notes = pasted; // Transfer ownership directly into Model
}

如需写入多格式剪贴板数据（文件列表、RGBA FluxBitmap 位图或自定义剪贴板格式），可通过 flux_data_create() 构造 FluxData * 并传给 flux_clipboard_set 。

切换主题、窗口材质与窗口状态¶

借助 on_start 保存的 m->app ，可在 update() 中响应设置变更并实时调整窗口外观：

  • 深浅色主题 ： flux_theme_setmode(flux_app_theme(m->app), FLUX_THEME_DARK); （可选 FLUX_THEME_SYSTEM 、 FLUX_THEME_LIGHT 或 FLUX_THEME_DARK ）。

  • 窗口背景材质 ： flux_window_setbackdrop(flux_app_window(m->app), FLUX_BACKDROP_MICA_ALT); （可选 FLUX_BACKDROP_NONE 、 MICA 、 MICA_ALT 或 ACRYLIC ）。

  • 动态窗口标题与关闭窗口 ： flux_window_settitle(flux_app_window(m->app), L"Unsaved *"); 与 flux_window_close(flux_app_window(m->app)); （例如在确认退出 ContentDialog 后主动关闭窗口）。

后台工作线程通信¶

将网络请求、解压缩或重度索引任务放到后台工作线程中执行，使 update() 始终保持轻量响应。将 FluxApp * 传入工作线程上下文，即可通过线程安全的消息投递函数实时回报进度：

static DWORD WINAPI worker_proc(void * param) {
    FluxApp *app = param;
    for (int step = 0; step < 100; step++) {
        // Perform background work...
        flux_app_post(app, xtk_msg_i(MSG_PROGRESS_STEP, step + 1));
    }
    char const *summary = "All 100 items processed.";
    flux_app_poststring(app, xtk_msg(MSG_WORKER_LOG), summary, (uint32_t) strlen(summary));
    flux_app_post(app, xtk_msg(MSG_WORKER_DONE));
    return 0;
}

  • flux_app_post 可从任意线程将标量 XtkMsg 消息入队并唤醒 UI 循环。

  • flux_app_poststring 安全拷贝 UTF-8 字符串并在 UI 线程的 update() 中作为 (char const *) msg.ptr 送达。

  • 请在 FluxAppCfg.on_stop 中等待活动工作线程汇合退出后再结束 FluxApp 。完整的多页安装向导（含暂停、取消与后台线程同步）可参考 fluxent/examples/installer_wizard/ 。
